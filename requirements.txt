# ============================================================================
#  УСТАНОВКА llama-cpp-python НА WINDOWS
# ============================================================================
#
#  Вариант A — CPU (ничего собирать не нужно, работает всегда):
#      pip install llama-cpp-python==0.3.16
#
#  Вариант B — ГОТОВЫЙ CUDA-WHEEL (рекомендуется, компилятор НЕ нужен).
#  Подберите индекс под свою версию CUDA Toolkit (cu121 / cu122 / cu124):
#      pip install llama-cpp-python==0.3.16 ^
#          --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu124
#
#  Вариант C — сборка из исходников с CUDA (только при наличии компилятора):
#  Требуется: Visual Studio 2022 Build Tools (C++ workload) + CUDA Toolkit + CMake.
#      set CMAKE_ARGS=-DGGML_CUDA=on
#      set FORCE_CMAKE=1
#      pip install llama-cpp-python==0.3.16 --no-binary llama-cpp-python
#
#  Вариант D — сборка с OpenBLAS (ускорение на CPU без GPU):
#      set CMAKE_ARGS=-DGGML_BLAS=ON -DGGML_BLAS_VENDOR=OpenBLAS
#      pip install llama-cpp-python==0.3.16 --no-binary llama-cpp-python
#
#  После установки CUDA-сборки не забудьте в settings.json выставить
#  local_model.n_gpu_layers = -1 (все слои на GPU) или, например, 20.
#  При n_gpu_layers = 0 модель считается на CPU.
